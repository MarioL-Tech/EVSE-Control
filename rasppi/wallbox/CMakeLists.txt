cmake_minimum_required(VERSION 3.16)
project(evse_wallbox LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_WALLBOX_SERVICE "Build service (requires libmodbus)" ON)

add_library(wallbox_data src/telemetry.cpp)
target_include_directories(wallbox_data PUBLIC src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(wallbox_data PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_WALLBOX_SERVICE)
    find_package(Threads REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBMODBUS REQUIRED IMPORTED_TARGET libmodbus>=3.1.6)
    pkg_check_modules(LIBMOSQUITTO REQUIRED IMPORTED_TARGET libmosquitto>=2.0)
    add_library(wallbox_write src/write_core.cpp)
    target_include_directories(wallbox_write PUBLIC src)
    target_link_libraries(wallbox_write PUBLIC wallbox_data PkgConfig::LIBMODBUS Threads::Threads)
    add_library(wallbox_control src/control_channel.cpp)
    target_link_libraries(wallbox_control PUBLIC wallbox_write PkgConfig::LIBMOSQUITTO)
    add_executable(evse-wallbox src/main.cpp src/mqtt.cpp)
    target_link_libraries(evse-wallbox PRIVATE wallbox_control)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(wallbox_write PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(wallbox_control PRIVATE -Wall -Wextra -Wpedantic)
        target_compile_options(evse-wallbox PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    install(TARGETS evse-wallbox RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(wallbox-tests tests/telemetry_test.cpp)
    target_link_libraries(wallbox-tests PRIVATE wallbox_data)
    add_test(NAME wallbox-telemetry COMMAND wallbox-tests)
    if(BUILD_WALLBOX_SERVICE)
        add_executable(wallbox-control-tests tests/control_channel_test.cpp)
        target_link_libraries(wallbox-control-tests PRIVATE wallbox_control)
        add_test(NAME wallbox-control-protocol COMMAND wallbox-control-tests)
        set_tests_properties(wallbox-control-protocol PROPERTIES TIMEOUT 10)
        add_executable(wallbox-write-tests tests/write_core_test.cpp)
        target_link_libraries(wallbox-write-tests PRIVATE wallbox_write)
        add_test(NAME wallbox-write-validation COMMAND wallbox-write-tests)
        set_tests_properties(wallbox-write-validation PROPERTIES TIMEOUT 10)
    endif()
    if(BUILD_WALLBOX_SERVICE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_executable(wallbox-write-probe tests/write_probe.cpp)
            target_link_libraries(wallbox-write-probe PRIVATE wallbox_write)
            if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
                target_compile_options(wallbox-write-tests PRIVATE -Wall -Wextra -Wpedantic)
                target_compile_options(wallbox-write-probe PRIVATE -Wall -Wextra -Wpedantic)
            endif()
            add_test(NAME wallbox-write-simulation
                COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/write_test.py
                        $<TARGET_FILE:wallbox-write-probe>)
            set_tests_properties(wallbox-write-simulation PROPERTIES TIMEOUT 90)
            add_test(NAME wallbox-rtu-simulation
                COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/rtu_test.py
                        $<TARGET_FILE:evse-wallbox>)
            set_tests_properties(wallbox-rtu-simulation PROPERTIES TIMEOUT 20)
            find_program(MOSQUITTO_BROKER mosquitto)
            find_program(MOSQUITTO_SUB mosquitto_sub)
            if(MOSQUITTO_BROKER AND MOSQUITTO_SUB)
                add_test(NAME wallbox-mqtt-integration
                    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mqtt_test.py
                            $<TARGET_FILE:evse-wallbox>)
                set_tests_properties(wallbox-mqtt-integration PROPERTIES TIMEOUT 45)
            endif()
        endif()
    endif()
endif()
