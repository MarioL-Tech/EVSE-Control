cmake_minimum_required(VERSION 3.16)
project(evse_wallbox LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_WALLBOX_SERVICE "Build service (requires libmodbus)" ON)

add_library(wallbox_data src/telemetry.cpp)
target_include_directories(wallbox_data PUBLIC src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(wallbox_data PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_WALLBOX_SERVICE)
    find_package(Threads REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LIBMODBUS REQUIRED IMPORTED_TARGET libmodbus>=3.1.6)
    pkg_check_modules(LIBMOSQUITTO REQUIRED IMPORTED_TARGET libmosquitto>=2.0)
    add_executable(evse-wallbox src/main.cpp src/mqtt.cpp)
    target_link_libraries(evse-wallbox PRIVATE wallbox_data PkgConfig::LIBMODBUS PkgConfig::LIBMOSQUITTO Threads::Threads)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(evse-wallbox PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    install(TARGETS evse-wallbox RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(wallbox-tests tests/telemetry_test.cpp)
    target_link_libraries(wallbox-tests PRIVATE wallbox_data)
    add_test(NAME wallbox-telemetry COMMAND wallbox-tests)
    if(BUILD_WALLBOX_SERVICE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_test(NAME wallbox-rtu-simulation
                COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/rtu_test.py
                        $<TARGET_FILE:evse-wallbox>)
            set_tests_properties(wallbox-rtu-simulation PROPERTIES TIMEOUT 20)
            find_program(MOSQUITTO_BROKER mosquitto)
            find_program(MOSQUITTO_SUB mosquitto_sub)
            if(MOSQUITTO_BROKER AND MOSQUITTO_SUB)
                add_test(NAME wallbox-mqtt-integration
                    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mqtt_test.py
                            $<TARGET_FILE:evse-wallbox>)
                set_tests_properties(wallbox-mqtt-integration PROPERTIES TIMEOUT 45)
            endif()
        endif()
    endif()
endif()
